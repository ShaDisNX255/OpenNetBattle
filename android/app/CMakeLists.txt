# For more information about using CMake with Android Studio, read the
# documentation: https://d.android.com/studio/projects/add-native-code.html

# Sets the minimum version of CMake required to build the native library.

cmake_minimum_required(VERSION 3.14)
project(battlenetwork_native LANGUAGES C CXX)

# ---- C++ standard ------------------------------------------------------------
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-g -fno-omit-frame-pointer)


# ---- Paths to externals ------------------------------------------------------
set(SFML_REPO ${CMAKE_CURRENT_SOURCE_DIR}/../../extern/SFML)
set(LUA_SRC_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../extern/lua/src)
set(SFMIDI_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../../BattleNetwork/sfMidi)

# Poco installed prefix (generated by your "cmake --build ... --target install")
# Example: extern/poco/dist-android-arm64-v8a
set(POCO_PREFIX ${CMAKE_CURRENT_SOURCE_DIR}/../../extern/poco/dist-android-${ANDROID_ABI})

# ---- Prebuilt third‑party (headers + libs staged by SFML) --------------------
set(OPENAL_LIBRARY   ${SFML_REPO}/extlibs/libs-android/${ANDROID_ABI}/libopenal.so)
set(FREETYPE_LIBRARY ${SFML_REPO}/extlibs/libs-android/${ANDROID_ABI}/libfreetype.a)
set(VORBISENC_LIBRARY ${SFML_REPO}/extlibs/libs-android/${ANDROID_ABI}/libvorbisenc.a)
set(VORBISFILE_LIBRARY ${SFML_REPO}/extlibs/libs-android/${ANDROID_ABI}/libvorbisfile.a)
set(VORBIS_LIBRARY    ${SFML_REPO}/extlibs/libs-android/${ANDROID_ABI}/libvorbis.a)
set(OGG_LIBRARY       ${SFML_REPO}/extlibs/libs-android/${ANDROID_ABI}/libogg.a)
set(FLAC_LIBRARY      ${SFML_REPO}/extlibs/libs-android/${ANDROID_ABI}/libFLAC.a)

# Optional: headers if you include them directly
include_directories(${SFML_REPO}/extlibs/headers/freetype2 ${SFML_REPO}/extlibs/headers)

# ---- Definitions -------------------------------------------------------------
# Required by SFML on Android
add_definitions(-DSFML_OPENGL_ES2)

# When statically linking Poco, define this
# (You built static libs in dist-android-<abi>/lib/)
add_definitions(-DPOCO_STATIC)

# ---- Tell SFML where deps are (so it doesn't run find_package on host) -------
# Use CACHE so these propagate into the SFML subproject.
set(OPENAL_INCLUDE_DIR   ${SFML_REPO}/extlibs/headers/AL                                  CACHE PATH     "OpenAL include")
set(OPENAL_LIBRARY       ${SFML_REPO}/extlibs/libs-android/${ANDROID_ABI}/libopenal.so    CACHE FILEPATH "OpenAL library")

set(FREETYPE_LIBRARY     ${SFML_REPO}/extlibs/libs-android/${ANDROID_ABI}/libfreetype.a   CACHE FILEPATH "FreeType static lib")

set(OGG_INCLUDE_DIR      ${SFML_REPO}/extlibs/headers/ogg                                 CACHE PATH     "Ogg include")
set(OGG_LIBRARY          ${SFML_REPO}/extlibs/libs-android/${ANDROID_ABI}/libogg.a        CACHE FILEPATH "Ogg static lib")

set(VORBIS_INCLUDE_DIR   ${SFML_REPO}/extlibs/headers/vorbis                              CACHE PATH     "Vorbis include")
set(VORBIS_LIBRARY       ${SFML_REPO}/extlibs/libs-android/${ANDROID_ABI}/libvorbis.a     CACHE FILEPATH "Vorbis static lib")
set(VORBISFILE_LIBRARY   ${SFML_REPO}/extlibs/libs-android/${ANDROID_ABI}/libvorbisfile.a CACHE FILEPATH "Vorbisfile static lib")
set(VORBISENC_LIBRARY    ${SFML_REPO}/extlibs/libs-android/${ANDROID_ABI}/libvorbisenc.a  CACHE FILEPATH "Vorbisenc static lib")

set(FLAC_INCLUDE_DIR     ${SFML_REPO}/extlibs/headers/flac                                CACHE PATH     "FLAC include")
set(FLAC_LIBRARY         ${SFML_REPO}/extlibs/libs-android/${ANDROID_ABI}/libFLAC.a       CACHE FILEPATH "FLAC static lib")


# Optional: quick echo to verify during configure
message(STATUS "ANDROID_ABI=${ANDROID_ABI}")
message(STATUS "OPENAL_INCLUDE_DIR=${OPENAL_INCLUDE_DIR}")
message(STATUS "OPENAL_LIBRARY=${OPENAL_LIBRARY}")
message(STATUS "POCO_PREFIX=${POCO_PREFIX}")

add_compile_definitions(BN_MOD_SUPPORT)

set(CMAKE_DEBUG_POSTFIX "" CACHE STRING "" FORCE)
# ---- Build SFML as subproject (gives sfml-* targets) -------------------------
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../extern/SFML ./sfml-build)

add_library(fluidsynth SHARED IMPORTED)
set_target_properties(fluidsynth PROPERTIES
        IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/../../extern/fluidsynth/build/src/libfluidsynth.so"
        INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/../../extern/fluidsynth/build/include;${CMAKE_CURRENT_LIST_DIR}/../../extern/fluidsynth/include"
)

# ---- Your sources ------------------------------------------------------------
file(GLOB_RECURSE BATTLENETWORK_SRC
        ${CMAKE_CURRENT_SOURCE_DIR}/../../BattleNetwork/*.cpp
        ${CMAKE_CURRENT_SOURCE_DIR}/../../BattleNetwork/Android/*.cpp)

file(GLOB_RECURSE bnFiles CONFIGURE_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/../../BattleNetwork/*.h"
        "${CMAKE_CURRENT_SOURCE_DIR}/../../BattleNetwork/*.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/../../BattleNetwork/netplay/*.h"
        "${CMAKE_CURRENT_SOURCE_DIR}/../../BattleNetwork/netplay/*.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/../../BattleNetwork/overworld/*.h"
        "${CMAKE_CURRENT_SOURCE_DIR}/../../BattleNetwork/overworld/*.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/../../BattleNetwork/widgets/*.h"
        "${CMAKE_CURRENT_SOURCE_DIR}/../../BattleNetwork/widgets/*.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/../../BattleNetwork/bindings/*.h"
        "${CMAKE_CURRENT_SOURCE_DIR}/../../BattleNetwork/bindings/*.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/../../BattleNetwork/stx/*.h"
        "${CMAKE_CURRENT_SOURCE_DIR}/../../BattleNetwork/stx/*.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/../../BattleNetwork/zip/*.h"
        "${CMAKE_CURRENT_SOURCE_DIR}/../../BattleNetwork/zip/*.c"
        "${CMAKE_CURRENT_SOURCE_DIR}/../../BattleNetwork/crypto/*.h"
        "${CMAKE_CURRENT_SOURCE_DIR}/../../BattleNetwork/sago/*.h"
        "${CMAKE_CURRENT_SOURCE_DIR}/../../BattleNetwork/sago/*.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/../../BattleNetwork/mmbn.ico.c"
)



# ---- JNI shared lib (this becomes libnative-lib.so) --------------------------
add_library(native-lib SHARED
        ${BATTLENETWORK_SRC}
        ${bnFiles}
        src/main/cpp/native-lib.cpp
)

file(GLOB_RECURSE SFMIDI_SOURCES
        ${SFMIDI_ROOT}/src/*.cpp
)

# ---- Includes for your target ------------------------------------------------
target_include_directories(native-lib PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/../../extern/Swoosh/src
        ${CMAKE_CURRENT_SOURCE_DIR}/../../BattleNetwork
        ${CMAKE_CURRENT_SOURCE_DIR}/../../BattleNetwork/sfMidi/include
        ${CMAKE_CURRENT_SOURCE_DIR}/../../BattleNetwork/sfMidi/src
        ${CMAKE_CURRENT_SOURCE_DIR}/../../BattleNetwork/sol/include
        ${CMAKE_CURRENT_SOURCE_DIR}/../../extern/fluidsynth/build/include
        ${CMAKE_CURRENT_SOURCE_DIR}/../../extern/fluidsynth/include
        ${CMAKE_CURRENT_SOURCE_DIR}/../../extern/SFML/src
        ${CMAKE_CURRENT_SOURCE_DIR}/src/main/cpp/native-lib.cpp
        # Use the installed Poco include root
        ${POCO_PREFIX}/include
)

# If you prefer target-level definitions instead of add_definitions:
# target_compile_definitions(native-lib PRIVATE SFML_OPENGL_ES2 POCO_STATIC)

# ---- NDK system libs ---------------------------------------------------------
find_library(LOG_LIB log)
find_library(ANDROID_LIB android)
find_library(EGL_LIB EGL)
find_library(GLES2_LIB GLESv2)
find_library(OPENSLES_LIB OpenSLES)



add_library(lua STATIC
        ${LUA_SRC_DIR}/lapi.c
        ${LUA_SRC_DIR}/lcode.c
        ${LUA_SRC_DIR}/lctype.c
        ${LUA_SRC_DIR}/ldebug.c
        ${LUA_SRC_DIR}/ldo.c
        ${LUA_SRC_DIR}/ldump.c
        ${LUA_SRC_DIR}/lfunc.c
        ${LUA_SRC_DIR}/lgc.c
        ${LUA_SRC_DIR}/llex.c
        ${LUA_SRC_DIR}/lmem.c
        ${LUA_SRC_DIR}/lobject.c
        ${LUA_SRC_DIR}/lopcodes.c
        ${LUA_SRC_DIR}/lparser.c
        ${LUA_SRC_DIR}/lstate.c
        ${LUA_SRC_DIR}/lstring.c
        ${LUA_SRC_DIR}/ltable.c
        ${LUA_SRC_DIR}/ltm.c
        ${LUA_SRC_DIR}/lundump.c
        ${LUA_SRC_DIR}/lvm.c
        ${LUA_SRC_DIR}/lzio.c
        ${LUA_SRC_DIR}/lauxlib.c
        ${LUA_SRC_DIR}/lbaselib.c
        ${LUA_SRC_DIR}/lcorolib.c
        ${LUA_SRC_DIR}/ldblib.c
        ${LUA_SRC_DIR}/liolib.c
        ${LUA_SRC_DIR}/lmathlib.c
        ${LUA_SRC_DIR}/loadlib.c
        ${LUA_SRC_DIR}/loslib.c
        ${LUA_SRC_DIR}/lstrlib.c
        ${LUA_SRC_DIR}/ltablib.c
        ${LUA_SRC_DIR}/lutf8lib.c
)
# Export Lua headers so your code can include <lua.h>
target_include_directories(lua PUBLIC ${LUA_SRC_DIR})

add_library(sfmidi STATIC ${SFMIDI_SOURCES})
target_include_directories(sfmidi PRIVATE
        ${CMAKE_CURRENT_LIST_DIR}/../../extern/SFML/include
        ${CMAKE_CURRENT_LIST_DIR}/../../extern/fluidsynth/build/include
        ${CMAKE_CURRENT_LIST_DIR}/../../extern/fluidsynth/include
)
target_include_directories(sfmidi PUBLIC ${SFMIDI_ROOT}/include)

# Android system libs (often needed together with audio code)
find_library(log-lib log)
find_library(android-lib android)
find_library(opensles-lib OpenSLES)

# ---- Link everything to native-lib ------------------------------------------
target_link_libraries(native-lib
        sfmidi
        fluidsynth
        # NDK
        ${LOG_LIB}
        ${ANDROID_LIB}
        ${EGL_LIB}
        ${GLES2_LIB}
        ${OPENSLES_LIB}

        # SFML built from subdir
        sfml-system
        sfml-window
        sfml-audio
        sfml-graphics
        sfml-network
        sfml-activity

        # Pull in sfml-main objects if you rely on them
        -Wl,--whole-archive sfml-main -Wl,--no-whole-archive

        # Poco (installed static libs)
        ${POCO_PREFIX}/lib/libPocoFoundation.a
        ${POCO_PREFIX}/lib/libPocoUtil.a
        ${POCO_PREFIX}/lib/libPocoNet.a

        # Audio stack (prebuilt for Android)
        ${FREETYPE_LIBRARY}
        ${VORBISENC_LIBRARY}
        ${VORBISFILE_LIBRARY}
        ${VORBIS_LIBRARY}
        ${OGG_LIBRARY}
        ${FLAC_LIBRARY}
        ${OPENAL_LIBRARY}
        lua
)

add_custom_command(TARGET native-lib POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${CMAKE_CURRENT_LIST_DIR}/../../extern/fluidsynth/build/src/libfluidsynth.so"
        "${CMAKE_LIBRARY_OUTPUT_DIRECTORY}/libfluidsynth.so"
)
